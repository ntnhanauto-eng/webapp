// Lấy thông tin cấu hình từ bảng app_settings (nếu chưa có sẽ lấy giá trị mặc định)
$settings = [
    'kid1_name'   => 'Trâm Anh',
    'kid1_class'  => '6A3',
    'kid1_pass'   => '123456',
    'kid2_name'   => 'Thành Phát',
    'kid2_class'  => '10A4',
    'kid2_pass'   => '123456',
    'admin_pass'  => 'admin123',
    'semester'    => 'HK1',
    'school_year' => '2026-2027'
];
try {
    $st_set = $conn->query("SELECT setting_key, setting_val FROM app_settings");
    if ($st_set) {
        $db_settings = $st_set->fetchAll(PDO::FETCH_KEY_PAIR);
        $settings = array_merge($settings, $db_settings);
    }
} catch (Exception $e) {
    // Sử dụng mặc định nếu bảng chưa tồn tại
}

// Xác định thứ hiện tại (0 = CN, 1 = T2, ..., 6 = T7)
$current_w = (int)date('w');
$today_field = ($current_w == 0) ? 'cn' : 'thu' . ($current_w + 1);

// Lấy thông tin tháng và năm hiện tại
$current_month_str = 'Tháng ' . date('n / Y');

// Tính ngày tháng của tuần hiện tại (Thứ 2 -> Chủ Nhật)
$monday_timestamp = strtotime('this week monday');
$week_dates = [];
$days_keys = ['thu2', 'thu3', 'thu4', 'thu5', 'thu6', 'thu7', 'cn'];
$days_vn_short = ['T2', 'T3', 'T4', 'T5', 'T6', 'T7', 'CN'];
$days_vn_full  = ['Thứ Hai', 'Thứ Ba', 'Thứ Tư', 'Thứ Năm', 'Thứ Sáu', 'Thứ Bảy', 'Chủ Nhật'];

for ($i = 0; $i < 7; $i++) {
    $ts = strtotime("+$i days", $monday_timestamp);
    $week_dates[$days_keys[$i]] = [
        'short'     => $days_vn_short[$i],
        'full_name' => $days_vn_full[$i],
        'date'      => date('d', $ts),
        'month'     => date('m', $ts),
        'full_date' => date('d/m', $ts)
    ];
}

// Lấy danh bạ giáo viên lưu vào mảng Map theo ID để tra cứu tức thì
$teachers_map = [];
$teachers_by_kid = ['1' => [], '2' => []];
try {
    $stmt_tc = $conn->query("SELECT * FROM teachers ORDER BY be_name ASC, subject_name ASC");
    $all_teachers = $stmt_tc->fetchAll(PDO::FETCH_ASSOC);
    foreach ($all_teachers as $tc) {
        $teachers_map[$tc['id']] = $tc;
        $teachers_by_kid[$tc['be_name']][] = $tc;
    }
} catch (Exception $e) {
    $teachers_map = [];
}

// Lấy dữ liệu Thời Khóa Biểu
$stmt = $conn->query("SELECT * FROM schedule ORDER BY FIELD(buoi, 'SÁNG', 'CHIỀU', 'TỐI'), id ASC");
$raw_schedules = $stmt->fetchAll(PDO::FETCH_ASSOC);

// Chuyển dữ liệu sang mảng App
$app_data = [];
foreach ($raw_schedules as $row) {
    $be_id = (string)$row['be_name'];
    foreach ($days_keys as $dk) {
        $raw_content = trim($row[$dk] ?? '');
        if (!empty($raw_content) && $raw_content !== '-') {
            $lines = array_values(array_filter(explode("\n", str_replace("\r", "", $raw_content))));
            $subject = trim($lines[0] ?? '');
            $note = isset($lines[1]) ? trim(implode(' ', array_slice($lines, 1))) : '';

            $teacher_id = $row['t_' . $dk] ?? null;
            $teacher_info = ($teacher_id && isset($teachers_map[$teacher_id])) ? $teachers_map[$teacher_id] : null;

            $app_data[] = [
                'id'          => $row['id'],
                'be_id'       => $be_id,
                'day'         => $dk,
                'buoi'        => $row['buoi'],
                'tiet'        => $row['tiet'],
                'subject'     => $subject,
                'note'        => $note,
                'teacher'     => $teacher_info
            ];
        }
    }
}

        async function checkRainAlert() {
            try {
                const res = await fetch('https://api.open-meteo.com/v1/forecast?latitude=21.0285&longitude=105.8542&hourly=precipitation,precipitation_probability', { cache: 'no-store' });
                if (res.ok) {
                    const data = await res.json();
                    const now = new Date();
                    const currentHour = now.getHours();
                    let rainStartTime = null;
                    let rainEndTime = null;
                    
                    if (data.hourly && data.hourly.precipitation && data.hourly.time) {
                        for (let i = 0; i < data.hourly.time.length; i++) {
                            const forecastTime = new Date(data.hourly.time[i]);
                            // Quét khung giờ chiều từ 15:00 đến 18:00
                            if (forecastTime.getDate() === now.getDate() && forecastTime.getHours() >= 15 && forecastTime.getHours() <= 18) {
                                if (data.hourly.precipitation[i] > 0.2 || (data.hourly.precipitation_probability && data.hourly.precipitation_probability[i] > 60)) {
                                    if (!rainStartTime) {
                                        rainStartTime = String(forecastTime.getHours()).padStart(2, '0') + ':00';
                                    }
                                    // Giả định cơn mưa kéo dài qua mỗi mốc giờ có dữ liệu mưa
                                    rainEndTime = String(forecastTime.getHours() + 1).padStart(2, '0') + ':00';
                                }
                            }
                        }
                    }

                    const alertBox = document.getElementById('rainAlertBox');
                    const alertText = document.getElementById('rainAlertText');
                    if (alertBox && alertText) {
                        if (rainStartTime && currentHour < 18) {
                            alertText.innerHTML = `<b>Cảnh báo mưa:</b> Dự báo có mưa tại khu vực trường học vào khoảng từ <b>${rainStartTime} đến ${rainEndTime}</b>. Nhớ chuẩn bị ô hoặc áo mưa cho con!`;
                            alertBox.style.display = 'flex';
                        } else {
                            alertBox.style.display = 'none';
                        }
                    }
                }
            } catch (e) {
                // Bỏ qua nếu mất kết nối
            }
        }

        function saveTodoTag() {
            const key = document.getElementById('todoSlotKey').value;
            const kidId = document.getElementById('todoKidId').value;
            const content = document.getElementById('todoContentInput').value.trim();
            const type = document.getElementById('todoTypeSelect').value;
            const expireType = document.getElementById('todoExpireSelect').value;
            const enteredPass = document.getElementById('todoPasswordInput').value.trim();

            if (!content) {
                alert('Vui lòng nhập nội dung ghi chú!');
                return;
            }

            if (!enteredPass) {
                alert('Vui lòng nhập mật khẩu xác nhận!');
                document.getElementById('todoPasswordInput').focus();
                return;
            }

            if (!verifyKidOrAdminPassword(kidId, enteredPass)) {
                alert('Mật khẩu không chính xác! Hãy nhập mật khẩu của đúng bé này hoặc mật khẩu Admin.');
                document.getElementById('todoPasswordInput').focus();
                return;
            }

            const now = new Date();
            let expireAt = 0;

            if (expireType === 'today_end') {
                const endToday = new Date(now.getFullYear(), now.getMonth(), now.getDate(), 23, 59, 59);
                expireAt = endToday.getTime();
            } else if (expireType === '24h') {
                expireAt = Date.now() + 24 * 60 * 60 * 1000;
            } else if (expireType === '3d') {
                expireAt = Date.now() + 3 * 24 * 60 * 60 * 1000;
            } else if (expireType === 'weekend') {
                const day = now.getDay();
                const distToSun = (7 - day) % 7;
                const sunday = new Date(now.getFullYear(), now.getMonth(), now.getDate() + distToSun, 23, 59, 59);
                expireAt = sunday.getTime();
            }

            const tags = getTodoTags();
            tags[key] = {
                type: type,
                content: content,
                expireAt: expireAt
            };

            localStorage.setItem('tkb_todo_tags', JSON.stringify(tags));
            syncTodoTagsToCloud(tags);
            closeTodoTagModal();
            renderAgenda();
        }

        function toggleVoiceSpeech() {
            if (!('speechSynthesis' in window)) {
                alert('Rất tiếc, trình duyệt của bạn không hỗ trợ tính năng đọc bằng giọng nói.');

            const speechText = generateDailySpeechText();
            if (!speechText) {
                alert('Không có dữ liệu thời khóa biểu để đọc.');
                return;
            }

            window.speechSynthesis.cancel();

            const utterance = new SpeechSynthesisUtterance(speechText);
            utterance.lang = 'vi-VN';
            utterance.rate = 1.0;
            utterance.pitch = 1.0;

            const voices = window.speechSynthesis.getVoices();
            const viVoice = voices.find(v => v.lang.includes('vi') || v.lang.includes('VIE'));
            if (viVoice) {
                utterance.voice = viVoice;
            }

            utterance.onstart = () => {
                setSpeakingUI(true);
            };

            utterance.onend = () => {
                setSpeakingUI(false);
            };

            utterance.onerror = () => {
                setSpeakingUI(false);
            };

            window.speechSynthesis.speak(utterance);
        }

        function generateDailySpeechText() {
            const dayInfo = weekDates[currentDay];
            let list = scheduleData.filter(i => i.day === currentDay);
            const todoTags = getTodoTags();

            if (list.length === 0) {
                return `${dayInfo.full_name}, ngày ${dayInfo.date} tháng ${dayInfo.month}. Hôm nay không có lịch học. Chúc các bé nghỉ ngơi vui vẻ!`;
            }

            let text = `Thời khóa biểu ${dayInfo.full_name}, ngày ${dayInfo.date} tháng ${dayInfo.month}. `;

            const readForKid = (kId) => {
                const isKid1 = (kId === '1');
                const kidName = isKid1 ? (appSettings.kid1_name || 'Trâm Anh') : (appSettings.kid2_name || 'Thành Phát');
                let kidList = list.filter(i => i.be_id === kId);

                if (kidList.length === 0) {
                    return `Bé ${kidName} hôm nay không có tiết học nào. `;
                }

                kidList.sort((a, b) => getItemStartMinutes(a) - getItemStartMinutes(b));

                let kidText = `Lịch học của bé ${kidName} có ${kidList.length} tiết: `;

                let buoiGroups = {};
                kidList.forEach(item => {
                    let b = item.buoi || 'SÁNG';
                    if (!buoiGroups[b]) buoiGroups[b] = [];
                    buoiGroups[b].push(item);
                });

                for (let b in buoiGroups) {
                    kidText += `Buổi ${b.toLowerCase()}: `;
                    buoiGroups[b].forEach(item => {
                        const cleanTiet = (item.tiet || '').split('(')[0].trim();
                        const slot = parseSlotMinutes(item);
                        const timeStartStr = slot ? slot.timeRangeStr.split('-')[0].trim() : '';
                        
                        kidText += `${cleanTiet} môn ${item.subject} lúc ${timeStartStr}. `;

                        const slotKey = `${item.id}_${item.be_id}_${item.day}`;
                        if (todoTags[slotKey]) {
                            kidText += `Lưu ý có ${todoTags[slotKey].content}. `;
                        }
                    });
                }
                return kidText;
            };

            if (currentProfile === 'all') {
                text += readForKid('1') + ' ' + readForKid('2');
            } else {
                text += readForKid(currentProfile);
            }

            return text;
        }

        function getEstimatedTimeRange(buoi, tiet) {
            const num = parseInt((tiet || '').replace(/\D/g, '')) || 1;
            const b = (buoi || '').trim().toUpperCase();
            if (b.includes('SÁNG') || b.includes('SANG')) {
                const times = ['07:15 - 08:00', '08:05 - 08:50', '09:05 - 09:50', '10:00 - 10:45', '10:50 - 11:35'];
                return times[num - 1] || '07:30 - 11:15';
            } else if (b.includes('CHIỀU') || b.includes('CHIEU')) {
                const times = ['13:30 - 14:15', '14:20 - 15:05', '15:20 - 16:05', '16:10 - 16:55', '17:00 - 17:45'];
                return times[num - 1] || '13:30 - 17:00';
            }
            return '18:00 - 19:30';
        }

        function timeToMinutes(h, m) {
            return parseInt(h, 10) * 60 + parseInt(m, 10);
        }

        function parseSlotMinutes(item) {
            const customText = (item.note || '') + ' ' + (item.subject || '');
            let match = customText.match(/(\d{1,2})[:hH](\d{2})\s*[~-]\s*(\d{1,2})[:hH](\d{2})/);
            
            if (!match) {
                match = (item.tiet || '').match(/(\d{1,2})[:hH](\d{2})\s*[~-]\s*(\d{1,2})[:hH](\d{2})/);
            }
            
            if (!match) {
                const defaultRange = getEstimatedTimeRange(item.buoi, item.tiet);
                match = defaultRange.match(/(\d{1,2})[:hH](\d{2})\s*[~-]\s*(\d{1,2})[:hH](\d{2})/);
            }

            if (match) {
                let start = timeToMinutes(match[1], match[2]);
                let end = timeToMinutes(match[3], match[4]);
                let timeRangeStr = match[1].padStart(2, '0') + ':' + match[2] + ' - ' + match[3].padStart(2, '0') + ':' + match[4];
                return { start, end, timeRangeStr };
            }

            return null;
        }

        function getItemStartMinutes(item) {
            const slot = parseSlotMinutes(item);
            if (slot) return slot.start;

            const b = (item.buoi || '').trim().toUpperCase();
            let base = 7 * 60;
            if (b.includes('CHIỀU') || b.includes('CHIEU')) {
                base = 13 * 60 + 30;
            } else if (b.includes('TỐI') || b.includes('TOI')) {
                base = 18 * 60;
            }

            const match = (item.tiet || '').match(/(?:tiết\s*|^)(\d+)/i);
            const num = match ? parseInt(match[1]) : 1;
            return base + (num - 1) * 50;
        }

                const numA = matchA ? parseInt(matchA[1]) : 1;
                const matchB = (b.tiet || '').match(/(?:tiết\s*|^)(\d+)/i);
                const numB = matchB ? parseInt(matchB[1]) : 1;
                if (numA !== numB) return numA - numB;

                return (parseInt(a.be_id) || 1) - (parseInt(b.be_id) || 1);
            });

            const currentMinutes = now.getHours() * 60 + now.getMinutes();

        function formatRemainingTime(minutesDiff) {
            if (minutesDiff <= 0) return 'bây giờ';

            return `${h}h ${m}p`;
        }

            let list = scheduleData.filter(i => i.day === realTodayKey && i.be_id === kidId);

            if (list.length === 0) {
                return {
                    status: 'NGHỈ NGƠI',
                    statusBg: 'rgba(148, 163, 184, 0.2)',
                    statusColor: '#94a3b8',
                    subject: 'Không có lịch học hôm nay',
                    child: kidName,
                    childColor: kidColor,
                    time: 'Thảnh thơi 🎉',
                    timeHint: 'Không có tiết nào',
                    icon: 'fa-mug-hot',
                    iconColor: '#94a3b8',
                    targetCardId: null,
                    isClickable: false
                };
            }

            let parsedList = list.map(item => {
                let range = parseSlotMinutes(item);
                let startMin = range ? range.start : getItemStartMinutes(item);
                let endMin = range ? range.end : (startMin + 45);
                return {
                    item: item,
                    start: startMin,
                    end: endMin,
                    timeRangeStr: range ? range.timeRangeStr : getEstimatedTimeRange(item.buoi, item.tiet)
                };
            });

            parsedList.sort((a, b) => a.start - b.start);

            // 1. ĐANG DIỄN RA
            let ongoing = parsedList.find(p => currentMinutes >= p.start && currentMinutes <= p.end);
            if (ongoing) {
                let cleanTiet = ongoing.item.tiet.split('(')[0].trim();
                let remaining = ongoing.end - currentMinutes;
                return {
                    status: 'ĐANG DIỄN RA',
                    statusBg: 'rgba(234, 88, 12, 0.25)',
                    statusColor: '#f97316',
                    subject: ongoing.item.subject,
                    child: kidName,
                    childColor: kidColor,
                    time: cleanTiet + ' (' + ongoing.timeRangeStr + ')',
                    timeHint: `Còn khoảng ${remaining} phút là tan tiết`,
                    icon: 'fa-book-open-reader',
                    iconColor: '#ea580c',
                    targetCardId: `slot-card-${ongoing.item.id}-${ongoing.item.be_id}-${ongoing.item.day}`,
                    isClickable: true
                };
            }

            // 2. TÌM MÔN HỌC KẾ TIẾP
            let nextSlot = parsedList.find(p => p.start > currentMinutes);

            if (nextSlot) {
                let diffMinutes = nextSlot.start - currentMinutes;
                let cleanTiet = nextSlot.item.tiet.split('(')[0].trim();
                let cardId = `slot-card-${nextSlot.item.id}-${nextSlot.item.be_id}-${nextSlot.item.day}`;

                if (diffMinutes <= 60) {
                    return {
                        status: 'SẮP DIỄN RA',
                        statusBg: 'rgba(245, 158, 11, 0.25)',
                        statusColor: '#fbbf24',
                        subject: nextSlot.item.subject,
                        child: kidName,
                        childColor: kidColor,
                        time: cleanTiet + ' (' + nextSlot.timeRangeStr + ')',
                        timeHint: `Bắt đầu sau ${diffMinutes} phút`,
                        icon: 'fa-hourglass-half',
                        iconColor: '#f59e0b',
                        targetCardId: cardId,
                        isClickable: true
                    };
                }

                let buoiText = (nextSlot.item.buoi || '').toUpperCase();
                let buoiLabel = buoiText.includes('CHIỀU') ? 'Buổi Chiều' : (buoiText.includes('TỐI') ? 'Buổi Tối' : 'Tiếp theo');
                let startTimeStr = nextSlot.timeRangeStr.split('-')[0].trim();

                return {
                    status: 'TIẾT TIẾP THEO',
                    statusBg: 'rgba(59, 130, 246, 0.22)',
                    statusColor: '#60a5fa',
                    subject: nextSlot.item.subject,
                    child: kidName,
                    childColor: kidColor,
                    time: `${buoiLabel} lúc ${startTimeStr}`,
                    timeHint: `Còn khoảng ${formatRemainingTime(diffMinutes)} nữa`,
                    icon: 'fa-calendar-clock',
                    iconColor: '#38bdf8',
                    targetCardId: null,
                    isClickable: false
                };
            }

            // 4. ĐÃ HOÀN THÀNH TẤT CẢ TIẾT TRONG NGÀY
            return {
                status: 'ĐÃ HOÀN THÀNH',
                statusBg: 'rgba(16, 185, 129, 0.22)',
                statusColor: '#34d399',
                subject: 'Đã hoàn thành các môn hôm nay',
                child: kidName,
                childColor: kidColor,
                time: 'Đã hết tiết học',
                timeHint: 'Nghỉ ngơi & chuẩn bị bài mới',
                icon: 'fa-circle-check',
                iconColor: '#10b981',
                targetCardId: null,
                isClickable: false
            };
        }

        function scrollToSubjectCard(cardId) {
            if (!cardId) return;

            if (currentDay !== realTodayKey) {
                changeDay(realTodayKey);
            }

            setTimeout(() => {
                const target = document.getElementById(cardId);
                if (target) {
                    target.scrollIntoView({ behavior: 'smooth', block: 'center' });
                    
                    target.classList.add('card-highlight-flash');
                    setTimeout(() => {
                        target.classList.remove('card-highlight-flash');
                    }, 1800);
                }
            }, 100);
        }

        function updateDynamicWidget() {
            const widget = document.getElementById('upcomingWidget');

            if (currentDay !== realTodayKey) {
                widget.style.display = 'none';
                return;
            }

            widget.style.display = 'block';

            const now = new Date();
            const currentMinutes = now.getHours() * 60 + now.getMinutes();

            let targetKids = (currentProfile === 'all') ? ['1', '2'] : [currentProfile];
            let html = '';
            targetKids.forEach(kidId => {
                let info = getStatusForKid(kidId, currentMinutes);
                let clickAttr = info.isClickable && info.targetCardId ? `onclick="scrollToSubjectCard('${info.targetCardId}')"` : '';
                let cursorClass = info.isClickable ? 'clickable' : '';
                let navIcon = info.isClickable ? `<i class="fa-solid fa-arrow-turn-down" style="font-size:10px; margin-left:4px; opacity:0.8;"></i>` : '';

                html += `
                    <div class="widget-item ${cursorClass}" ${clickAttr} title="${info.isClickable ? 'Bấm để xem vị trí môn học bên dưới' : ''}">
                        <div class="widget-left">
                            <div class="widget-icon"><i class="fa-solid ${info.icon}" style="color: ${info.iconColor};"></i></div>
                            <div style="min-width:0;">
                                <span class="widget-status-badge" style="background: ${info.statusBg}; color: ${info.statusColor};">
                                    ${info.status} ${navIcon}
                                </span>
                                <div class="widget-subject">${info.subject}</div>
                            </div>
                        </div>
                        <div class="widget-right">
                            <div class="widget-child" style="color: ${info.childColor};">${info.child}</div>
                            <div class="widget-time">${info.time}</div>
                            <div class="widget-time-hint">${info.timeHint}</div>
                        </div>
                    </div>
                `;
            });

            const rainBoxHtml = document.getElementById('rainAlertBox').outerHTML;
            widget.innerHTML = rainBoxHtml + html;
        }

        // VUỐT MÀN HÌNH ĐỔI NGÀY TRÊN MOBILE
        let touchStartX = 0;
        let touchStartY = 0;
        let touchEndX = 0;
        let touchEndY = 0;

        document.addEventListener('touchstart', function(e) {
            if (window.matchMedia("(orientation: landscape)").matches || window.innerWidth >= 1024 || e.target.closest('#dateStrip') || e.target.closest('.modal-content') || e.target.closest('#backpackCard')) {
                touchStartX = 0;
                touchStartY = 0;
                return;
            }
            touchStartX = e.changedTouches[0].screenX;
            touchStartY = e.changedTouches[0].screenY;
        }, { passive: true });

        document.addEventListener('touchend', function(e) {
            if (!touchStartX && !touchStartY) return;
            touchEndX = e.changedTouches[0].screenX;
            touchEndY = e.changedTouches[0].screenY;
            handleGlobalGesture();
            touchStartX = 0;
            touchStartY = 0;
        }, { passive: true });

        function handleGlobalGesture() {
            const diffX = touchEndX - touchStartX;
            const diffY = touchEndY - touchStartY;

            if (Math.abs(diffX) > Math.abs(diffY) && Math.abs(diffX) > 45) {
                const currentIndex = daysKeysList.indexOf(currentDay);
                if (diffX < 0) {
                    if (currentIndex < daysKeysList.length - 1) {
                        changeDay(daysKeysList[currentIndex + 1]);
                    }
                } else {
                    if (currentIndex > 0) {
                        changeDay(daysKeysList[currentIndex - 1]);
                    }
                }
            }
        }

            renderLandscapeGrid();
        });
        window.addEventListener('orientationchange', () => {
            setTimeout(renderLandscapeGrid, 100);
        });

        // TỰ ĐỘNG DỌN SẠCH CACHE CŨ
        if ('caches' in window) {
            caches.keys().then((names) => {
                for (let name of names) {
                    if (name !== 'tkb-live-v1' && name !== 'tkb-static-v1') {
                        caches.delete(name);
                    }
                }
            });
        }

        if ('serviceWorker' in navigator) {
            window.addEventListener('load', () => {
                navigator.serviceWorker.register('sw.js').then((reg) => {
                    reg.update();
                }).catch(() => {});
            });
        }

        // ĐỒNG BỘ DỮ LIỆU ĐỘNG TỪ API
        async function syncDataFromServer() {
            if (!navigator.onLine) {
                return false;
            }
            try {
                const res = await fetch('api.php?t=' + Date.now(), { 
                    cache: 'no-store',
                    headers: { 'Pragma': 'no-cache', 'Cache-Control': 'no-cache' }
                });
                if (res.ok) {
                    const data = await res.json();
                    if (data && data.schedule_data) {
                        scheduleData = data.schedule_data;
                        rawSchedules = data.raw_schedules;
                        teachersMap = data.teachers_map;
                        appSettings = data.settings;

                        if (data.todo_tags !== undefined && typeof data.todo_tags === 'object') {
                            localStorage.setItem('tkb_todo_tags', JSON.stringify(data.todo_tags));
                        }

                        saveToOfflineStorage();
                        updateHeaderNames();

                        renderSubjectFilters();
                        renderBackpackAssistant();
                        renderAgenda();
                        updateDynamicWidget();
                        renderLandscapeGrid();
                        return true;
                    }
                }
            } catch (err) {
                console.log('Đang chạy ở chế độ offline hoặc mạng chậm.');
            }
            return false;
        }

        // NÚT LÀM MỚI BẰNG TAY
        async function manualSync() {
            const btn = document.getElementById('refreshBtn');
            const icon = btn.querySelector('i');
            icon.classList.add('fa-spin');
            
            if (navigator.onLine) {
                await syncDataFromServer();
            } else {
                showNetworkToast('offline', 'Bạn đang offline, không thể cập nhật trực tuyến');
            }
            
            setTimeout(() => {
                icon.classList.remove('fa-spin');
            }, 600);
        }

        document.addEventListener('visibilitychange', () => {
            if (document.visibilityState === 'visible' && navigator.onLine) {
                syncDataFromServer();
            }
        });

        // Khởi chạy ban đầu
        changeDay(realTodayKey);
        renderSubjectFilters();
        renderBackpackAssistant();
        renderLandscapeGrid();
        checkRainAlert();
        if (navigator.onLine) {
            syncDataFromServer();
        }

        // Tự động kiểm tra cập nhật mỗi 15 giây
        setInterval(() => {
            renderAgenda();
            updateDynamicWidget();
            if (navigator.onLine) {
                syncDataFromServer();
            }
        }, 15000);
