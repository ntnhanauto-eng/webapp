// Lấy thông tin cấu hình từ bảng app_settings (nếu chưa có sẽ lấy giá trị mặc định)
$settings = [
    'kid1_name'   => 'Trâm Anh',
    'kid1_class'  => '6A3',
    'kid1_pass'   => '123456',
    'kid2_name'   => 'Thành Phát',
    'kid2_class'  => '10A4',
    'kid2_pass'   => '123456',
    'admin_pass'  => 'admin123',
    'semester'    => 'HK1',
    'school_year' => '2026-2027'
];
try {
    $st_set = $conn->query("SELECT setting_key, setting_val FROM app_settings");
    if ($st_set) {
        $db_settings = $st_set->fetchAll(PDO::FETCH_KEY_PAIR);
        $settings = array_merge($settings, $db_settings);
    }
} catch (Exception $e) {
    // Sử dụng mặc định nếu bảng chưa tồn tại
}

// Xác định thứ hiện tại (0 = CN, 1 = T2, ..., 6 = T7)
$current_w = (int)date('w');
$today_field = ($current_w == 0) ? 'cn' : 'thu' . ($current_w + 1);

// Lấy thông tin tháng và năm hiện tại
$current_month_str = 'Tháng ' . date('n / Y');

// Tính ngày tháng của tuần hiện tại (Thứ 2 -> Chủ Nhật)
$monday_timestamp = strtotime('this week monday');
$week_dates = [];
$days_keys = ['thu2', 'thu3', 'thu4', 'thu5', 'thu6', 'thu7', 'cn'];
$days_vn_short = ['T2', 'T3', 'T4', 'T5', 'T6', 'T7', 'CN'];
$days_vn_full  = ['Thứ Hai', 'Thứ Ba', 'Thứ Tư', 'Thứ Năm', 'Thứ Sáu', 'Thứ Bảy', 'Chủ Nhật'];

for ($i = 0; $i < 7; $i++) {
    $ts = strtotime("+$i days", $monday_timestamp);
    $week_dates[$days_keys[$i]] = [
        'short'     => $days_vn_short[$i],
        'full_name' => $days_vn_full[$i],
        'date'      => date('d', $ts),
        'month'     => date('m', $ts),
        'full_date' => date('d/m', $ts)
    ];
}

// Lấy danh bạ giáo viên lưu vào mảng Map theo ID để tra cứu tức thì
$teachers_map = [];
$teachers_by_kid = ['1' => [], '2' => []];
try {
    $stmt_tc = $conn->query("SELECT * FROM teachers ORDER BY be_name ASC, subject_name ASC");
    $all_teachers = $stmt_tc->fetchAll(PDO::FETCH_ASSOC);
    foreach ($all_teachers as $tc) {
        $teachers_map[$tc['id']] = $tc;
        $teachers_by_kid[$tc['be_name']][] = $tc;
    }
} catch (Exception $e) {
    $teachers_map = [];
}

// Lấy dữ liệu Thời Khóa Biểu
$stmt = $conn->query("SELECT * FROM schedule ORDER BY FIELD(buoi, 'SÁNG', 'CHIỀU', 'TỐI'), id ASC");
$raw_schedules = $stmt->fetchAll(PDO::FETCH_ASSOC);

// Chuyển dữ liệu sang mảng App
$app_data = [];
foreach ($raw_schedules as $row) {
    $be_id = (string)$row['be_name'];
    foreach ($days_keys as $dk) {
        $raw_content = trim($row[$dk] ?? '');
        if (!empty($raw_content) && $raw_content !== '-') {
            $lines = array_values(array_filter(explode("\n", str_replace("\r", "", $raw_content))));
            $subject = trim($lines[0] ?? '');
            $note = isset($lines[1]) ? trim(implode(' ', array_slice($lines, 1))) : '';

            $teacher_id = $row['t_' . $dk] ?? null;
            $teacher_info = ($teacher_id && isset($teachers_map[$teacher_id])) ? $teachers_map[$teacher_id] : null;

            $app_data[] = [
                'id'          => $row['id'],
                'be_id'       => $be_id,
                'day'         => $dk,
                'buoi'        => $row['buoi'],
                'tiet'        => $row['tiet'],
                'subject'     => $subject,
                'note'        => $note,
                'teacher'     => $teacher_info
            ];
        }
    }
}

        function refreshRealTimeDates() {
            const now = new Date();
            const jsDay = now.getDay();
            const calculatedTodayKey = (jsDay === 0) ? 'cn' : 'thu' + (jsDay + 1);
            
            const diffToMonday = (jsDay === 0 ? -6 : 1) - jsDay;
            const mondayDate = new Date(now.getFullYear(), now.getMonth(), now.getDate() + diffToMonday);

            const daysVnShort = ['T2', 'T3', 'T4', 'T5', 'T6', 'T7', 'CN'];
            const daysVnFull  = ['Thứ Hai', 'Thứ Ba', 'Thứ Tư', 'Thứ Năm', 'Thứ Sáu', 'Thứ Bảy', 'Chủ Nhật'];

            const newWeekDates = {};
            for (let i = 0; i < 7; i++) {
                let d = new Date(mondayDate.getFullYear(), mondayDate.getMonth(), mondayDate.getDate() + i);
                let dayKey = daysKeysList[i];
                let dd = String(d.getDate()).padStart(2, '0');
                let mm = String(d.getMonth() + 1).padStart(2, '0');
                newWeekDates[dayKey] = {
                    short: daysVnShort[i],
                    full_name: daysVnFull[i],
                    date: dd,
                    month: mm,
                    full_date: `${dd}/${mm}`
                };

                let chipNum = document.getElementById('dnum-' + dayKey);
                if (chipNum) chipNum.innerText = dd;
            }

            weekDates = newWeekDates;
            
            const monthElem = document.getElementById('currentMonthStrDisplay');
            if (monthElem) {
                monthElem.innerText = `Tháng ${now.getMonth() + 1} / ${now.getFullYear()}`;
            }

            if (realTodayKey !== calculatedTodayKey) {
                realTodayKey = calculatedTodayKey;
                currentDay = realTodayKey;
                changeDay(realTodayKey);
            }
        }

        // --- 2. DỰ BÁO THỜI TIẾT GIỜ TAN TRƯỜNG NINH HÒA ---
        async function checkRainAlert() {
            const alertBox = document.getElementById('rainAlertBox');
            if (!alertBox) return;

            try {
                if (currentDay !== realTodayKey) {
                    alertBox.style.display = 'none';
                    return;
                }

                let targetList = scheduleData.filter(i => i.day === realTodayKey);
                if (currentProfile !== 'all') {
                    targetList = targetList.filter(i => i.be_id === currentProfile);
                }

                if (targetList.length === 0) {
                    alertBox.style.display = 'none';
                    return;
                }

                let morningStartMin = 24 * 60;
                let morningDismissMin = 0;
                let afternoonStartMin = 24 * 60;
                let afternoonDismissMin = 0;

                targetList.forEach(item => {
                    let slot = parseSlotMinutes(item);
                    if (!slot) return;
                    let b = (item.buoi || '').toUpperCase();
                    if (b.includes('SÁNG') || b.includes('SANG')) {
                        if (slot.start < morningStartMin) morningStartMin = slot.start;
                        if (slot.end > morningDismissMin) morningDismissMin = slot.end;
                    } else if (b.includes('CHIỀU') || b.includes('CHIEU')) {
                        if (slot.start < afternoonStartMin) afternoonStartMin = slot.start;
                        if (slot.end > afternoonDismissMin) afternoonDismissMin = slot.end;
                    }
                });

                const now = new Date();
                const currentMinNow = now.getHours() * 60 + now.getMinutes();

                const showMorningAlert = (morningDismissMin > 0) && 
                                         (currentMinNow >= (morningStartMin - 60)) && 
                                         (currentMinNow <= (morningDismissMin + 30));

                const showAfternoonAlert = (afternoonDismissMin > 0) && 
                                           (currentMinNow >= (afternoonStartMin - 60)) && 
                                           (currentMinNow <= (afternoonDismissMin + 30));

                if (!showMorningAlert && !showAfternoonAlert) {
                    alertBox.style.display = 'none';
                    return;
                }

                if (!res.ok) return;

                const data = await res.json();

                function evaluateWeatherAt(dismissMinutes) {
                    const dismissHour = Math.floor(dismissMinutes / 60);
                    const dismissMinute = dismissMinutes % 60;
                    const timeStr = String(dismissHour).padStart(2, '0') + ':' + String(dismissMinute).padStart(2, '0');

                    let rain = false;
                    let hot = false;
                    let temp = 30;

                    if (data.hourly && data.hourly.time) {
                        for (let i = 0; i < data.hourly.time.length; i++) {
                            const forecastTime = new Date(data.hourly.time[i]);
                            if (forecastTime.getDate() === now.getDate()) {
                                let fHour = forecastTime.getHours();
                                if (fHour >= (dismissHour - 1) && fHour <= dismissHour) {
                                    let precip = data.hourly.precipitation ? data.hourly.precipitation[i] : 0;
                                    let prob = data.hourly.precipitation_probability ? data.hourly.precipitation_probability[i] : 0;
                                    temp = data.hourly.temperature_2m ? Math.round(data.hourly.temperature_2m[i]) : 30;

                                    if (precip > 0.1 || prob > 50) {
                                        rain = true;
                                        break;
                                    } else if (temp >= 33) {
                                        hot = true;
                                    }
                                }
                            }
                        }
                    }
                    return { timeStr, rain, hot, temp };
                }

                let alertRowsHtml = '';

                if (showMorningAlert) {
                    let w = evaluateWeatherAt(morningDismissMin);
                    if (w.rain) {
                        alertRowsHtml += `
                            <div class="weather-alert-row alert-rain">
                                <i class="fa-solid fa-cloud-showers-heavy" style="font-size: 13px; flex-shrink: 0;"></i>

                            </div>
                        `;
                    } else if (w.hot) {
                        alertRowsHtml += `
                            <div class="weather-alert-row alert-hot">
                                <i class="fa-solid fa-sun" style="font-size: 13px; flex-shrink: 0;"></i>
                                <span><b>Tan học sáng (${w.timeStr}):</b> Ninh Hòa nắng nóng (${w.temp}°C). Nhắc con đội nón mũ và mặc áo khoác!</span>
                            </div>
                        `;
                    } else {
                        alertRowsHtml += `
                            <div class="weather-alert-row alert-fine">
                                <i class="fa-solid fa-cloud-sun" style="font-size: 13px; flex-shrink: 0;"></i>
                                <span><b>Tan học sáng (${w.timeStr}):</b> Thời tiết Ninh Hòa dịu mát (${w.temp}°C), thuận tiện đón con.</span>
                            </div>
                        `;
                    }
                }

                if (showAfternoonAlert) {
                    let w = evaluateWeatherAt(afternoonDismissMin);
                    if (w.rain) {
                        alertRowsHtml += `
                            <div class="weather-alert-row alert-rain">
                                <i class="fa-solid fa-cloud-showers-heavy" style="font-size: 13px; flex-shrink: 0;"></i>
                                <span><b>Tan học chiều (${w.timeStr}):</b> Dự báo Ninh Hòa có mưa (${w.temp}°C). Ba mẹ nhớ mang áo mưa đón con!</span>
                            </div>
                        `;
                    } else if (w.hot) {
                        alertRowsHtml += `
                            <div class="weather-alert-row alert-hot">
                                <i class="fa-solid fa-sun" style="font-size: 13px; flex-shrink: 0;"></i>
                                <span><b>Tan học chiều (${w.timeStr}):</b> Ninh Hòa còn nắng gắt (${w.temp}°C). Nhắc con đội nón mũ che nắng!</span>
                            </div>
                        `;
                    } else {
                        alertRowsHtml += `
                            <div class="weather-alert-row alert-fine">
                                <i class="fa-solid fa-cloud-sun" style="font-size: 13px; flex-shrink: 0;"></i>
                                <span><b>Tan học chiều (${w.timeStr}):</b> Trời mát mẻ (${w.temp}°C), thích hợp đón con tan trường.</span>
                            </div>
                        `;
                    }
                }

                alertBox.innerHTML = alertRowsHtml;
                alertBox.style.display = alertRowsHtml ? 'block' : 'none';

            } catch (e) {
                // Ignore network errors
            }

        function getTodoTags() {
            try {
                const raw = localStorage.getItem('tkb_todo_tags');
                let data = raw ? JSON.parse(raw) : {};
                if (!data || typeof data !== 'object' || Array.isArray(data)) {
                    data = {};
                }
                const now = Date.now();
                let hasChange = false;
                
                for (const key in data) {
                    if (data[key] && data[key].expireAt && now > data[key].expireAt) {
                        delete data[key];
                        hasChange = true;
                    }
                }
                if (hasChange) {
                    localStorage.setItem('tkb_todo_tags', JSON.stringify(data));
                }
                return data;
            } catch (e) {
                return {};
            }
        }

        async function syncTodoTagsToCloud(tags) {
            if (!navigator.onLine) return;
            try {
                const safeTags = (tags && typeof tags === 'object' && !Array.isArray(tags)) ? tags : {};

                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ todo_tags: safeTags })
                });
            } catch (e) {
                console.log('Lưu cloud ngầm: offline hoặc server bận');
            }
        }

        function saveTodoTag() {
            const key = document.getElementById('todoSlotKey').value;
            const kidId = document.getElementById('todoKidId').value;
            const content = document.getElementById('todoContentInput').value.trim();
            const type = document.getElementById('todoTypeSelect').value;
            const expireType = document.getElementById('todoExpireSelect').value;
            const enteredPass = document.getElementById('todoPasswordInput').value.trim();

            if (!content) {
                alert('Vui lòng nhập nội dung ghi chú!');
                return;
            }

            if (!enteredPass) {
                alert('Vui lòng nhập mật khẩu tài khoản của bé hoặc Admin để lưu!');
                document.getElementById('todoPasswordInput').focus();
                return;
            }

            if (!verifyKidOrAdminPassword(kidId, enteredPass)) {
                alert('Mật khẩu không chính xác! Bạn phải nhập mật khẩu của đúng bé này hoặc mật khẩu Admin để lưu.');
                document.getElementById('todoPasswordInput').focus();
                return;
            }

            const now = new Date();
            let expireAt = 0;

            if (expireType === 'today_end') {
                const endToday = new Date(now.getFullYear(), now.getMonth(), now.getDate(), 23, 59, 59);
                expireAt = endToday.getTime();
            } else if (expireType === '24h') {
                expireAt = Date.now() + 24 * 60 * 60 * 1000;
            } else if (expireType === '3d') {
                expireAt = Date.now() + 3 * 24 * 60 * 60 * 1000;
            } else if (expireType === 'weekend') {
                const day = now.getDay();
                const distToSun = (7 - day) % 7;
                const sunday = new Date(now.getFullYear(), now.getMonth(), now.getDate() + distToSun, 23, 59, 59);
                expireAt = sunday.getTime();
            }

            let tags = getTodoTags();
            if (!tags || typeof tags !== 'object' || Array.isArray(tags)) {
                tags = {};
            }

            tags[key] = {
                type: type,
                content: content,
                expireAt: expireAt
            };

            localStorage.setItem('tkb_todo_tags', JSON.stringify(tags));
            syncTodoTagsToCloud(tags);
            closeTodoTagModal();
            renderAgenda();
        }

        function toggleVoiceSpeech() {
            if (!('speechSynthesis' in window)) {
                alert('Rất tiếc, trình duyệt của bạn không hỗ trợ tính năng đọc bằng giọng nói.');

                setSpeakingUI(false);
                return;

            const speechText = generateDailySpeechText();
            if (!speechText) {
                alert('Không có dữ liệu thời khóa biểu để đọc.');
                return;
            }

            window.speechSynthesis.cancel();

            const utterance = new SpeechSynthesisUtterance(speechText);
            utterance.lang = 'vi-VN';
            utterance.rate = 1.0;
            utterance.pitch = 1.0;

            const voices = window.speechSynthesis.getVoices();
            const viVoice = voices.find(v => v.lang.includes('vi') || v.lang.includes('VIE'));
            if (viVoice) {
                utterance.voice = viVoice;
            }

            utterance.onstart = () => {
                setSpeakingUI(true);
            };

            utterance.onend = () => {
                setSpeakingUI(false);
            };

            utterance.onerror = () => {
                setSpeakingUI(false);
            };

            window.speechSynthesis.speak(utterance);
        }

        // --- XỬ LÝ THỜI GIAN ---
        function getEstimatedTimeRange(buoi, tiet) {
            const num = parseInt((tiet || '').replace(/\D/g, '')) || 1;
            const b = (buoi || '').trim().toUpperCase();
            if (b.includes('SÁNG') || b.includes('SANG')) {
                const times = ['07:15 - 08:00', '08:05 - 08:50', '09:05 - 09:50', '10:00 - 10:45', '10:50 - 11:35'];
                return times[num - 1] || '07:30 - 11:15';
            } else if (b.includes('CHIỀU') || b.includes('CHIEU')) {
                const times = ['13:30 - 14:15', '14:20 - 15:05', '15:20 - 16:05', '16:10 - 16:55', '17:00 - 17:45'];
                return times[num - 1] || '13:30 - 17:00';
            }
            return '18:00 - 19:30';
        }

        function timeToMinutes(h, m) {
            return parseInt(h, 10) * 60 + parseInt(m, 10);
        }

        function parseSlotMinutes(item) {
            const customText = (item.note || '') + ' ' + (item.subject || '');
            let match = customText.match(/(\d{1,2})[:hH](\d{2})\s*[~-]\s*(\d{1,2})[:hH](\d{2})/);
            
            if (!match) {
                match = (item.tiet || '').match(/(\d{1,2})[:hH](\d{2})\s*[~-]\s*(\d{1,2})[:hH](\d{2})/);
            }
            
            if (!match) {
                const defaultRange = getEstimatedTimeRange(item.buoi, item.tiet);
                match = defaultRange.match(/(\d{1,2})[:hH](\d{2})\s*[~-]\s*(\d{1,2})[:hH](\d{2})/);
            }

            if (match) {
                let start = timeToMinutes(match[1], match[2]);
                let end = timeToMinutes(match[3], match[4]);
                let timeRangeStr = match[1].padStart(2, '0') + ':' + match[2] + ' - ' + match[3].padStart(2, '0') + ':' + match[4];
                return { start, end, timeRangeStr };
            }

            return null;
        }

        function getItemStartMinutes(item) {
            const slot = parseSlotMinutes(item);
            if (slot) return slot.start;

            const b = (item.buoi || '').trim().toUpperCase();
            let base = 7 * 60;
            if (b.includes('CHIỀU') || b.includes('CHIEU')) {
                base = 13 * 60 + 30;
            } else if (b.includes('TỐI') || b.includes('TOI')) {
                base = 18 * 60;
            }

            const match = (item.tiet || '').match(/(?:tiết\s*|^)(\d+)/i);
            const num = match ? parseInt(match[1]) : 1;
            return base + (num - 1) * 50;
        }

                return;
            }

            list.sort((a, b) => {
                const startA = getItemStartMinutes(a);
                const startB = getItemStartMinutes(b);
                if (startA !== startB) return startA - startB;

                const matchA = (a.tiet || '').match(/(?:tiết\s*|^)(\d+)/i);
                const numA = matchA ? parseInt(matchA[1]) : 1;
                const matchB = (b.tiet || '').match(/(?:tiết\s*|^)(\d+)/i);
                const numB = matchB ? parseInt(matchB[1]) : 1;
                if (numA !== numB) return numA - numB;

                return (parseInt(a.be_id) || 1) - (parseInt(b.be_id) || 1);
            });
            
            const now = new Date();
            const currentMinutes = now.getHours() * 60 + now.getMinutes();
            const isViewingToday = (currentDay === realTodayKey);
            const todoTags = getTodoTags();

                    html += `<div class="buoi-divider"><span class="buoi-pill">BUỔI ${item.buoi}</span></div>`;

                        let formattedTime = (parts.length === 2) ? `${parts[0]}&nbsp;&ndash;&nbsp;${parts[1]}` : rawTime;

                        let startMin = slotRange ? slotRange.start : getItemStartMinutes(r);
                        let endMin = slotRange ? slotRange.end : (startMin + 45);
                        if (endMin <= startMin) endMin = startMin + 45;

                        standardSlots.push({
                            cleanTiet,
                            formattedTime,
                            startMin,
                            endMin
                        });
                    });

                    // Chiều cao tiêu chuẩn mỗi dòng tiết giảm xuống 50px để bảng TKB vừa khít 1 trang A4 ngang
                    const standardSlotHeight = 50;
                    const totalTrackHeight = standardSlots.length * standardSlotHeight;

                    html += `<tr>`;
                    // 1. Cột BUỔI
                    html += `<td class="col-buoi" style="height:${totalTrackHeight}px;">${buoiKey}</td>`;

                    // 2. Cột TIẾT
                    html += `<td class="col-tiet" style="padding:0; height:${totalTrackHeight}px;">`;
                    standardSlots.forEach((slot, idx) => {
                        let isLast = idx === standardSlots.length - 1;
                        html += `
                            <div class="landscape-time-col-cell" style="height:${standardSlotHeight}px; ${isLast ? 'border-bottom:none;' : ''}">
                                <b>${slot.cleanTiet}</b>
                                <div style="font-size:9px; color:#64748b; margin-top:1px; white-space:nowrap;">${slot.formattedTime}</div>
                            </div>
                        `;
                    });
                    html += `</td>`;

                    // 3. Các cột THỨ (T2 -> CN)
                    daysKeysList.forEach(dk => {
                        let dayItems = [];
                        rows.forEach((r, slotIndex) => {
                            let rawCell = trimString(r[dk] || '');
                            if (rawCell && rawCell !== '-') {
                                let lines = rawCell.split("\n").map(l => l.trim()).filter(l => l !== '');
                                let subject = lines[0] || '';
                                let note = lines.slice(1).join(' ').trim();
                                let teacherId = r['t_' + dk] || null;
                                let teacherObj = (teacherId && teachersMap[teacherId]) ? teachersMap[teacherId] : null;

                                let currentCellItem = {
                                    buoi: r.buoi,
                                    tiet: r.tiet,
                                    subject: subject,
                                    note: note
                                };
                                let cellSlotRange = parseSlotMinutes(currentCellItem);
                                let start = cellSlotRange ? cellSlotRange.start : getItemStartMinutes(r);
                                let end = cellSlotRange ? cellSlotRange.end : (start + 45);
                                if (end <= start) end = start + 45;

                                dayItems.push({
                                    id: r.id,
                                    be_id: kId,
                                    day: dk,
                                    subject: subject,
                                    note: note,
                                    teacher: teacherObj,
                                    start: start,
                                    end: end,
                                    slotIndex: slotIndex,
                                    slotKey: `${r.id}_${kId}_${dk}`
                                });
                            }
                        });

                        html += `<td style="height:${totalTrackHeight}px; position:relative; padding:0;">`;
                        html += `<div class="landscape-day-track" style="height:${totalTrackHeight}px;">`;

                        // Đường kẻ ngang ngăn cách từng tiết chuẩn
                        for (let s = 1; s < standardSlots.length; s++) {
                            html += `<div class="landscape-slot-grid-line" style="top:${s * standardSlotHeight}px;"></div>`;
                        }

                        // Hàm nội suy thời gian sang toạ độ Y chính xác theo từng hàng tiết
                        function getPixelForMinute(targetMin) {
                            if (standardSlots.length === 0) return 0;
                            if (targetMin <= standardSlots[0].startMin) return 0;
                            const lastIndex = standardSlots.length - 1;
                            if (targetMin >= standardSlots[lastIndex].endMin) return totalTrackHeight;

                            for (let i = 0; i < standardSlots.length; i++) {
                                const sSlot = standardSlots[i];
                                if (targetMin >= sSlot.startMin && targetMin <= sSlot.endMin) {
                                    const progress = (targetMin - sSlot.startMin) / (sSlot.endMin - sSlot.startMin || 45);
                                    return i * standardSlotHeight + progress * standardSlotHeight;
                                }
                                if (i < lastIndex && targetMin > sSlot.endMin && targetMin < standardSlots[i + 1].startMin) {
                                    return (i + 1) * standardSlotHeight;
                                }
                            }
                            return totalTrackHeight;
                        }

                        // Hiển thị các môn học: Giảm khoảng cách viền trên dưới tối đa
                        dayItems.forEach(item => {
                            const curStandard = standardSlots[item.slotIndex];
                            let topPx, heightPx;

                            const isAlignedWithStandard = curStandard && (Math.abs(item.start - curStandard.startMin) <= 3) && (Math.abs(item.end - curStandard.endMin) <= 3);

                            if (isAlignedWithStandard) {
                                topPx = item.slotIndex * standardSlotHeight + 2;
                                heightPx = standardSlotHeight - 4;
                            } else {
                                const rawTop = getPixelForMinute(item.start);
                                const rawBottom = getPixelForMinute(item.end);

                                topPx = rawTop + 2;
                                heightPx = (rawBottom - rawTop) - 4;

                                heightPx = Math.max(heightPx, 32);
                                if ((topPx + heightPx) > totalTrackHeight - 2) {
                                    heightPx = totalTrackHeight - topPx - 2;
                                }
                            }

                            let isCurrentSlot = false;
                            let isPastSlot = false;
                            if (dk === realTodayKey) {
                                if (currentMinutes >= item.start && currentMinutes <= item.end) {
                                    isCurrentSlot = true;
                                } else if (currentMinutes > item.end) {
                                    isPastSlot = true;
                                }
                            }

                            let safeSubEnc = encodeURIComponent(item.subject).replace(/'/g, "%27");
                            let teacherObjStr = item.teacher ? encodeURIComponent(JSON.stringify(item.teacher)).replace(/'/g, "%27") : '';

                            html += `
                                <div class="grid-sub-box ${isCurrentSlot ? 'is-current-slot' : ''} ${isPastSlot ? 'is-past-slot' : ''}" 
                                     style="top:${topPx}px; height:${heightPx}px;">
                                    <div class="grid-sub-title">${item.subject}</div>
                                    ${item.teacher ? `
                                        <div class="grid-sub-teacher" onclick="event.stopPropagation(); showTeacherDetails('${safeSubEnc}', '${teacherObjStr}')">
                                            <i class="fa-solid fa-user-tie" style="font-size:8px;"></i> ${item.teacher.teacher_name}
                                        </div>
                                    ` : ''}
                                    ${item.note ? `<div class="grid-sub-note"><i class="fa-solid fa-comment-dots" style="margin-right:2px; font-size:7.5px;"></i>${item.note}</div>` : ''}
                                </div>
                            `;
                        });

                        html += `</div></td>`;
                    });

                    html += `</tr>`;
                });

                html += `
                                </tbody>
                            </table>
                        </div>
                    </div>
                `;
            });

            container.innerHTML = html;
        }

        function formatRemainingTime(minutesDiff) {
            if (minutesDiff <= 0) return 'bây giờ';
            const h = Math.floor(minutesDiff / 60);
            const m = minutesDiff % 60;
            if (h === 0) return `${m} phút`;
            if (m === 0) return `${h} giờ`;
            return `${h}h ${m}p`;
        }

        function getStatusForKid(kidId, currentMinutes) {
            let kidName = (kidId === '1') ? (appSettings.kid1_name || 'Trâm Anh') : (appSettings.kid2_name || 'Thành Phát');
            let kidColor = (kidId === '1') ? '#10b981' : '#3b82f6';
            let list = scheduleData.filter(i => i.day === realTodayKey && i.be_id === kidId);

            if (list.length === 0) {
                return {
                    status: 'NGHỈ NGƠI',
                    statusBg: 'rgba(148, 163, 184, 0.2)',
                    statusColor: '#94a3b8',
                    subject: 'Không có lịch học hôm nay',
                    child: kidName,
                    childColor: kidColor,
                    time: 'Thảnh thơi 🎉',
                    timeHint: 'Không có tiết nào',
                    icon: 'fa-mug-hot',
                    iconColor: '#94a3b8',
                    targetCardId: null,
                    isClickable: false
                };
            }

            let parsedList = list.map(item => {
                let range = parseSlotMinutes(item);
                let startMin = range ? range.start : getItemStartMinutes(item);
                let endMin = range ? range.end : (startMin + 45);
                return {
                    item: item,
                    start: startMin,
                    end: endMin,
                    timeRangeStr: range ? range.timeRangeStr : getEstimatedTimeRange(item.buoi, item.tiet)
                };
            });

            parsedList.sort((a, b) => a.start - b.start);

            // 1. ĐANG DIỄN RA
            let ongoing = parsedList.find(p => currentMinutes >= p.start && currentMinutes <= p.end);
            if (ongoing) {
                let cleanTiet = ongoing.item.tiet.split('(')[0].trim();
                let remaining = ongoing.end - currentMinutes;
                return {
                    status: 'ĐANG DIỄN RA',
                    statusBg: 'rgba(234, 88, 12, 0.25)',
                    statusColor: '#f97316',
                    subject: ongoing.item.subject,
                    child: kidName,
                    childColor: kidColor,
                    time: cleanTiet + ' (' + ongoing.timeRangeStr + ')',
                    timeHint: `Còn khoảng ${remaining} phút là tan tiết`,
                    icon: 'fa-book-open-reader',
                    iconColor: '#ea580c',
                    targetCardId: `slot-card-${ongoing.item.id}-${ongoing.item.be_id}-${ongoing.item.day}`,
                    isClickable: true
                };
            }

            // 2. TÌM MÔN HỌC KẾ TIẾP
            let nextSlot = parsedList.find(p => p.start > currentMinutes);

            if (nextSlot) {
                let diffMinutes = nextSlot.start - currentMinutes;
                let cleanTiet = nextSlot.item.tiet.split('(')[0].trim();
                let cardId = `slot-card-${nextSlot.item.id}-${nextSlot.item.be_id}-${nextSlot.item.day}`;

                if (diffMinutes <= 60) {
                    return {
                        status: 'SẮP DIỄN RA',
                        statusBg: 'rgba(245, 158, 11, 0.25)',
                        statusColor: '#fbbf24',
                        subject: nextSlot.item.subject,
                        child: kidName,
                        childColor: kidColor,
                        time: cleanTiet + ' (' + nextSlot.timeRangeStr + ')',
                        timeHint: `Bắt đầu sau ${diffMinutes} phút`,
                        icon: 'fa-hourglass-half',
                        iconColor: '#f59e0b',
                        targetCardId: cardId,
                        isClickable: true
                    };
                }

                let buoiText = (nextSlot.item.buoi || '').toUpperCase();
                let buoiLabel = buoiText.includes('CHIỀU') ? 'Buổi Chiều' : (buoiText.includes('TỐI') ? 'Buổi Tối' : 'Tiếp theo');
                let startTimeStr = nextSlot.timeRangeStr.split('-')[0].trim();

                return {
                    status: 'TIẾT TIẾP THEO',
                    statusBg: 'rgba(59, 130, 246, 0.22)',
                    statusColor: '#60a5fa',
                    subject: nextSlot.item.subject,
                    child: kidName,
                    childColor: kidColor,
                    time: `${buoiLabel} lúc ${startTimeStr}`,
                    timeHint: `Còn khoảng ${formatRemainingTime(diffMinutes)} nữa`,
                    icon: 'fa-calendar-clock',
                    iconColor: '#38bdf8',
                    targetCardId: null,
                    isClickable: false
                };
            }

            // 3. ĐÃ HOÀN THÀNH TẤT CẢ TIẾT TRONG NGÀY
            return {
                status: 'ĐÃ HOÀN THÀNH',
                statusBg: 'rgba(16, 185, 129, 0.22)',
                statusColor: '#34d399',
                subject: 'Tất cả tiết học',
                child: kidName,
                childColor: kidColor,
                time: 'Đã hết tiết học',
                timeHint: 'Nghỉ ngơi & chuẩn bị bài mới',
                icon: 'fa-circle-check',
                iconColor: '#10b981',
                targetCardId: null,
                isClickable: false
            };
        }

        // --- VUỐT MÀN HÌNH ĐỔI NGÀY TRÊN MOBILE ---
        let touchStartX = 0;
        let touchStartY = 0;
        let touchEndX = 0;
        let touchEndY = 0;

        document.addEventListener('touchstart', function(e) {
            if (window.matchMedia("(orientation: landscape)").matches || window.innerWidth >= 1024 || e.target.closest('#dateStrip') || e.target.closest('.modal-content') || e.target.closest('#landscapeGridSection')) {
                touchStartX = 0;
                touchStartY = 0;
                return;
            }
            touchStartX = e.changedTouches[0].screenX;
            touchStartY = e.changedTouches[0].screenY;
        }, { passive: true });

        document.addEventListener('touchend', function(e) {
            if (!touchStartX && !touchStartY) return;
            touchEndX = e.changedTouches[0].screenX;
            touchEndY = e.changedTouches[0].screenY;
            handleGlobalGesture();
            touchStartX = 0;
            touchStartY = 0;
        }, { passive: true });

        function handleGlobalGesture() {
            const diffX = touchEndX - touchStartX;
            const diffY = touchEndY - touchStartY;

            if (Math.abs(diffX) > Math.abs(diffY) && Math.abs(diffX) > 45) {
                const currentIndex = daysKeysList.indexOf(currentDay);
                if (diffX < 0) {
                    if (currentIndex < daysKeysList.length - 1) {
                        changeDay(daysKeysList[currentIndex + 1]);
                    }
                } else {
                    if (currentIndex > 0) {
                        changeDay(daysKeysList[currentIndex - 1]);
                    }
                }
            }
        }

        function toggleFullscreen() {
            const icon = document.getElementById('fullScreenIcon');
            if (!document.fullscreenElement && !document.webkitFullscreenElement) {
                const docEl = document.documentElement;
                if (docEl.requestFullscreen) {
                    docEl.requestFullscreen().catch(() => {});
                } else if (docEl.webkitRequestFullscreen) {
                    docEl.webkitRequestFullscreen();
                }
                if (icon) icon.className = 'fa-solid fa-compress';
            } else {
                if (document.exitFullscreen) {
                    document.exitFullscreen().catch(() => {});
                } else if (document.webkitExitFullscreen) {
                    document.webkitExitFullscreen();
                }
                if (icon) icon.className = 'fa-solid fa-expand';
            }
        }

        document.addEventListener('fullscreenchange', () => {
            const icon = document.getElementById('fullScreenIcon');
            if (icon) {
                icon.className = document.fullscreenElement ? 'fa-solid fa-compress' : 'fa-solid fa-expand';
            }
        });

        // TỰ ĐỘNG DỌN SẠCH CACHE CŨ
        if ('caches' in window) {
            caches.keys().then((names) => {
                for (let name of names) {
                    if (name !== 'tkb-live-v1' && name !== 'tkb-static-v1') {
                        caches.delete(name);
                    }
                }
            });
        }

        // ĐĂNG KÝ SERVICE WORKER CHO OFFLINE
        if ('serviceWorker' in navigator) {
            window.addEventListener('load', () => {
                navigator.serviceWorker.register('sw.js').then((reg) => {
                    reg.update();
                }).catch(() => {});
            });
        }

        // ĐỒNG BỘ DỮ LIỆU ĐỘNG TỪ API
        async function syncDataFromServer() {
            if (!navigator.onLine) {
                return false;
            }
            try {
                const res = await fetch('api.php?t=' + Date.now(), { 
                    cache: 'no-store',
                    headers: { 'Pragma': 'no-cache', 'Cache-Control': 'no-cache' }
                });
                if (res.ok) {
                    const data = await res.json();
                    if (data && data.schedule_data) {
                        scheduleData = data.schedule_data;
                        rawSchedules = data.raw_schedules;
                        teachersMap = data.teachers_map;
                        appSettings = Object.assign({}, appSettings, data.settings);

                        if (data.todo_tags !== undefined && data.todo_tags !== null) {
                            let serverTags = data.todo_tags;
                            if (Array.isArray(serverTags) || typeof serverTags !== 'object') {
                                serverTags = {};
                            }
                            localStorage.setItem('tkb_todo_tags', JSON.stringify(serverTags));
                        }

                        if (data.exam_schedules !== undefined && data.exam_schedules !== null) {
                            localStorage.setItem('tkb_exam_schedules', JSON.stringify(data.exam_schedules));
                        }

                        saveToOfflineStorage();
                        updateHeaderNames();

                        refreshRealTimeDates();
                        renderAgenda();
                        updateDynamicWidget();
                        renderLandscapeGrid();
                        checkRainAlert();
                        updateExamBadgeCount();
                        return true;
                    }
                }
            } catch (err) {
                console.log('Đang chạy ở chế độ offline hoặc mạng chậm.');
            }
            return false;
        }

        // NÚT LÀM MỚI BẰNG TAY
        async function manualSync() {
            const btn = document.getElementById('refreshBtn');
            const icon = btn.querySelector('i');
            icon.classList.add('fa-spin');
            
            refreshRealTimeDates();
            if (navigator.onLine) {
                await syncDataFromServer();
            } else {
                showNetworkToast('offline', 'Bạn đang offline, không thể cập nhật trực tuyến');
            }
            
            setTimeout(() => {
                icon.classList.remove('fa-spin');
            }, 600);
        }

        document.addEventListener('visibilitychange', () => {
            if (document.visibilityState === 'visible') {
                refreshRealTimeDates();
                if (navigator.onLine) {
                    syncDataFromServer();
                }
            }
        });

        // Khởi chạy ứng dụng ban đầu
        initPcViewMode();
        refreshRealTimeDates();
        changeDay(realTodayKey);
        renderLandscapeGrid();
        checkRainAlert();
        updateExamBadgeCount();
        if (navigator.onLine) {
            syncDataFromServer();
        }

        // Quét thời gian thực mỗi 15 giây & cập nhật ngày nếu qua nửa đêm
        setInterval(() => {
            refreshRealTimeDates();
            renderAgenda();
            updateDynamicWidget();
            if (navigator.onLine) {
                syncDataFromServer();
            }
        }, 15000);

        // Định kỳ 15 phút cập nhật lại dự báo thời tiết Ninh Hòa
        setInterval(() => {
            checkRainAlert();
        }, 15 * 60 * 1000);
    </script>
