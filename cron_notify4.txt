// 2. CẤU HÌNH TELEGRAM (Điền đúng thông tin của bạn vào 2 dòng này)
define('TELEGRAM_TOKEN', '8987907075:AAEJFwpZoiH65oO0wh6lP2fNBbNhtv3-cHE');
define('TELEGRAM_CHAT_ID', '1733868980');
// Hàm gửi tin nhắn Telegram dùng Stream Context (Không gây lỗi 500 do thiếu thư viện cURL)
function sendTelegramMessage($message) {
    if (TELEGRAM_TOKEN === 'ĐIỀN_BOT_TOKEN_VÀO_ĐÂY' || TELEGRAM_CHAT_ID === 'ĐIỀN_CHAT_ID_VÀO_ĐÂY') {
        return false;
    }
    
    $url = "https://api.telegram.org/bot" . TELEGRAM_TOKEN . "/sendMessage";
    $postData = [
        'chat_id'    => TELEGRAM_CHAT_ID,
        'text'       => $message,
        'parse_mode' => 'HTML'
    ];

    $options = [
        'http' => [
            'header'  => "Content-Type: application/x-www-form-urlencoded\r\n",
            'method'  => 'POST',
            'content' => http_build_query($postData),
            'timeout' => 15,
        ],
        'ssl' => [
            'verify_peer'      => false,
            'verify_peer_name' => false,
        ]
    ];

    $context = stream_context_create($options);
    return @file_get_contents($url, false, $context);
}

// 3. LẤY CẤU HÌNH ỨNG DỤNG
$settings = [
    'kid1_name'  => 'Trâm Anh', 
    'kid1_class' => '6A3',
    'kid2_name'  => 'Thành Phát', 
    'kid2_class' => '10A4'
];

try {
    $st_set = $conn->query("SELECT setting_key, setting_val FROM app_settings");

    }

$mode = $_GET['type'] ?? '';
$now_minutes = (int)date('G') * 60 + (int)date('i');
$today_w = (int)date('w');
$today_field = ($today_w === 0) ? 'cn' : 'thu' . ($today_w + 1);

// Chuyển đổi chuỗi giờ thành số phút trong ngày
function timeStrToMinutes($timeStr) {
    if (preg_match('/(\d{1,2})[:hH](\d{2})/', $timeStr, $m)) {
        return (int)$m[1] * 60 + (int)$m[2];
    }
    return 0;
}

// Bóc tách giờ tan học an toàn
function parseSlotTimeSafe($item) {
    $customText = ($item['note'] ?? '') . ' ' . ($item['subject'] ?? '');
    if (preg_match('/(\d{1,2})[:hH](\d{2})\s*[~-]\s*(\d{1,2})[:hH](\d{2})/', $customText, $m)) {
        return ['start' => timeStrToMinutes($m[1] . ':' . $m[2]), 'end' => timeStrToMinutes($m[3] . ':' . $m[4])];
    }
    $buoi = strtoupper($item['buoi'] ?? '');
    if (strpos($buoi, 'SÁNG') !== false || strpos($buoi, 'SANG') !== false) {
        return ['start' => 435, 'end' => 695]; // 07:15 - 11:35
    } else {
        return ['start' => 810, 'end' => 1065]; // 13:30 - 17:45
    }
}

// --- TÁC VỤ 1: QUÉT TRƯỚC GIỜ TAN HỌC 60 PHÚT ---
if ($mode === 'check_dismissal') {
    try {
        $stmt = $conn->query("SELECT * FROM schedule");
        $all_rows = $stmt->fetchAll(PDO::FETCH_ASSOC);
    } catch (Exception $e) {
        die("Lỗi truy vấn cơ sở dữ liệu.");
    }

    $morning_dismiss_min = 0;
    $afternoon_dismiss_min = 0;

    foreach ($all_rows as $row) {
        $content = trim($row[$today_field] ?? '');
        if (empty($content) || $content === '-') continue;

        $slot = parseSlotTimeSafe($row);
        $buoi = strtoupper($row['buoi'] ?? '');

        if (strpos($buoi, 'SÁNG') !== false || strpos($buoi, 'SANG') !== false) {
            if ($slot['end'] > $morning_dismiss_min) $morning_dismiss_min = $slot['end'];
        } elseif (strpos($buoi, 'CHIỀU') !== false || strpos($buoi, 'CHIEU') !== false) {
            if ($slot['end'] > $afternoon_dismiss_min) $afternoon_dismiss_min = $slot['end'];
        }
    }

    // Nhắc trước giờ tan học sáng 60 phút (dung sai ±7 phút theo chu kỳ cron)
    if ($morning_dismiss_min > 0 && abs($now_minutes - ($morning_dismiss_min - 60)) <= 7) {
        $time_str = sprintf('%02d:%02d', floor($morning_dismiss_min / 60), $morning_dismiss_min % 60);
        $msg = "⏰ <b>NHẮC CHUẨN BỊ ĐÓN CON TAN HỌC SÁNG</b>\n";
        $msg .= "⏱️ Dự kiến tan trường lúc: <b>{$time_str}</b>\n";
        $msg .= "━━━━━━━━━━━━━━━━━━━━\n";
        $msg .= "🌤️ Ba mẹ nhớ chuẩn bị áo mưa hoặc nón cho các con nhé!";
        
        sendTelegramMessage($msg);
        echo "Đã gửi thông báo trước giờ tan học sáng thành công.";
        exit;
    }

    // Nhắc trước giờ tan học chiều 60 phút
    if ($afternoon_dismiss_min > 0 && abs($now_minutes - ($afternoon_dismiss_min - 60)) <= 7) {
        $time_str = sprintf('%02d:%02d', floor($afternoon_dismiss_min / 60), $afternoon_dismiss_min % 60);
        $msg = "⏰ <b>NHẮC CHUẨN BỊ ĐÓN CON TAN HỌC CHIỀU</b>\n";
        $msg .= "⏱️ Dự kiến tan trường lúc: <b>{$time_str}</b>\n";
        $msg .= "━━━━━━━━━━━━━━━━━━━━\n";
        $msg .= "🌤️ Ba mẹ nhớ chuẩn bị áo mưa hoặc nón cho các con nhé!";
        
        sendTelegramMessage($msg);
        echo "Đã gửi thông báo trước giờ tan học chiều thành công.";
        exit;
    }

    echo "Trạng thái: Chưa tới khung giờ cần nhắc nhở tan học.";
}
